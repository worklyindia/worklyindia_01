# Search for jobs
def search_jobs(query, location, page=1, limit=20):
    params = {
        'search_query': query,
        'location': location,
        'page': page,
        'limit': limit
    }
    
    response = requests.get(
        f'https://jubwzelupbntbetpajqb.supabase.co/functions/v1/jobs-search',
        headers=headers,
        params=params
    )
    
    response.raise_for_status()  # Raises an HTTPError for bad responses
    return response.json()

# Usage
jobs = search_jobs('developer', 'mumbai')
print(f'Found {len(jobs.get("data", []))} jobs')
for job in jobs.get('data', []):
    print(f'{job["title"]} at {job.get("company", {}).get("name", "Unknown")}')
Get Job Details by ID

Fetch complete details for a specific job:

Copy
# Get job details by ID
def get_job_details(job_id: str) -> dict:
    """
    Fetch complete job details including company info, skills, requirements
    
    Args:
        job_id: UUID of the job
        
    Returns:
        dict: Complete job data or None if not found
    """
    try:
        params = {'job_id': job_id}
        response = requests.get(
            f'https://jubwzelupbntbetpajqb.supabase.co/functions/v1/jobs-search',
            headers=headers,
            params=params
        )
        response.raise_for_status()
        data = response.json()
        
        if data.get('success') and len(data.get('data', [])) > 0:
            job = data['data'][0]
            print(f"Job Title: {job['title']}")
            print(f"Company: {job['company_name']}")
            print(f"Location: {job['location']}")
            print(f"Salary: {job['salary_currency']} {job['salary_min']} - {job['salary_max']}")
            print(f"Skills: {', '.join(job['skills'])}")
            print(f"Workly India: {job['remote_work']}")
            print(f"Applications: {job['application_count']}")
            return job
        else:
            print("Job not found")
            return None
            
    except requests.exceptions.HTTPError as e:
        print(f"HTTP Error: {e}")
        return None

# Usage
job_details = get_job_details('123e4567-e89b-12d3-a456-426614174000')

Error Handling

Handle API errors gracefully:

Copy
from requests.exceptions import HTTPError, RequestException

def api_call_with_error_handling():
    try:
        response = requests.get(
            f'https://jubwzelupbntbetpajqb.supabase.co/functions/v1/jobs-search',
            headers=headers,
            params={'search_query': 'developer'}
        )
        
        response.raise_for_status()
        return response.json()
        
    except HTTPError as e:
        if e.response.status_code == 429:
            print('Rate limit exceeded')
            retry_after = e.response.headers.get('Retry-After')
            print(f'Retry after: {retry_after} seconds')
        elif e.response.status_code == 401:
            print('Invalid API key')
        else:
            print(f'HTTP Error: {e.response.status_code}')
            print(f'Response: {e.response.text}')
        raise
        
    except RequestException as e:
        print(f'Request failed: {str(e)}')
        raise
Async Support (with aiohttp)

Install aiohttp: pip install aiohttp

Async example:

Copy
import aiohttp
import asyncio

async def search_jobs_async(query, location):
    async with aiohttp.ClientSession() as session:
        params = {'search_query': query, 'location': location}
        
        async with session.get(
            f'https://jubwzelupbntbetpajqb.supabase.co/functions/v1/jobs-search',
            headers=headers,
            params=params
        ) as response:
            response.raise_for_status()
            return await response.json()

async def create_multiple_jobs(jobs_data):
    async with aiohttp.ClientSession() as session:
        tasks = []
        
        for job_data in jobs_data:
            task = session.post(
                f'https://jubwzelupbntbetpajqb.supabase.co/functions/v1/jobs-create',
                headers=headers,
                json=job_data
            )
            tasks.append(task)
        
        responses = await asyncio.gather(*tasks)
        return [await r.json() for r in responses]

# Usage
async def main():
    jobs = await search_jobs_async('developer', 'mumbai')
    print(f'Found {len(jobs.get("data", []))} jobs')

asyncio.run(main())
Pagination Helper

Easily paginate through results:

Copy
def paginated_search(query, location, limit=50):
    """Generator function to iterate through all pages"""
    page = 1
    
    while True:
        response = requests.get(
            f'https://jubwzelupbntbetpajqb.supabase.co/functions/v1/jobs-search',
            headers=headers,
            params={
                'search_query': query,
                'location': location,
                'page': page,
                'limit': limit
            }
        )
        
        response.raise_for_status()
        data = response.json()
        
        jobs = data.get('data', [])
        if not jobs:
            break
            
        yield jobs
        
        # Check if there are more pages
        meta = data.get('meta', {})
        if page >= meta.get('total_pages', 1):
            break
            
        page += 1

# Usage
for page_jobs in paginated_search('developer', 'mumbai'):
    for job in page_jobs:
        print(f'Job: {job["title"]}')

Check the X-RateLimit-Remaining header in responses. When you get a 429 status code, wait for the time specified in the Retry-After header before making more requests.
Include your API key in the X-Api-Key header for all requests. Never expose your API key in client-side code.
Free tier allows 100 requests/day with 1 API key. Pro tier provides 10,000 requests/day with 2 API keys and priority support.
Paid actions (job posting, boosting, extension and interviews) are paid via Razorpay in the employer portal. Jobs created through the API are saved as drafts until payment is completed.
Boost packages increase your job's visibility. Package 1 is Standard (100 points/week), Package 2 is Premium (200 points/week), and Package 3 is Featured (300 points/week). The boost_level parameter (1-3) determines visibility priority.
If your wallet balance is insufficient, you'll receive a 402 Payment Required error. You'll need to add more points to your wallet before you can boost or extend jobs.
Use requests for simple synchronous applications. Use aiohttp for async/await support and better performance in concurrent scenarios.
